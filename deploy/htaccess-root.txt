# 入口（ルート）へ置く .htaccess の下書き。公開の直前に、`.htaccess` という名前で置く。
# 今の入口の .htaccess（2025/01/25 更新）は、控えを取ってから、置き換える。
#   今の内容：SVGの設定2行と、WordPressの標準の転送規則（重複して2回）だけ。独自の規則（転送、制限、認証）は、なし。
# /kdhs/ には、それ自身の .htaccess があり、入口の転送規則に頼っていないことを確認済み（2026-11-18）。
# ただし、入口の設定（DirectoryIndex、Header、ErrorDocument、Options など）は、/kdhs/ にも引き継がれる。変更のたびに、/kdhs/ の表示を確認すること。
# 試験用フォルダ（staging_new）で、画面の表示とコンソールのエラーを確認してから使う。

# 今の設定から、残すもの（SVG画像の設定）
AddType image/svg+xml .svg .svgz
AddEncoding gzip .svgz

# 入口は、静的な index.html を先に表示する。index.php も残す（/kdhs/ などの WordPress が、この設定を引き継ぐため。index.html だけにすると、/kdhs/ が表示されなくなる）
DirectoryIndex index.html index.php

ErrorDocument 404 /404.html
Options -Indexes
ServerSignature Off

# 暗号化されていない通信（http）を、https に転送する。
# 今までは、WordPress（Really Simple SSL）が、この役目をしていた可能性が高い。WordPressをやめると、なくなるため、ここで補う。
<IfModule mod_rewrite.c>
  RewriteEngine On
  # www 付きのアドレスは、www なし・https に、一度で転送する（今は、WordPressが行っているとみられる）
  RewriteCond %{HTTP_HOST} ^www\.kurume-dental\.or\.jp$ [NC]
  RewriteRule ^(.*)$ https://kurume-dental.or.jp/$1 [R=301,L]
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]
</IfModule>

# 旧WordPress時代のURLを、新しいURLへ、サーバー側で転送する（301）。
# 旧サイトの「Redirection」の規則のうち、会員向けを除く6件と、WordPressが自動で転送していた「推測URL」。
# 新サイトにも、同じ場所に、移動用の小さなページ（.html / .php）を置いてあるが、こちらの転送が、先に働く。
RedirectMatch 301 ^/business_checkup\.html$ /business_checkup/
RedirectMatch 301 ^/assistant_course\.php$ /assistant_course/
RedirectMatch 301 ^/event\.php$ /event/
RedirectMatch 301 ^/holiday_clinic\.php$ /holiday_clinic/
RedirectMatch 301 ^/local_dentist\.php$ /local_dentist_top/
RedirectMatch 301 ^/introduction\.html$ /introduction/
RedirectMatch 301 ^/business_checkup_([1-4])/?$ /business_checkup/business_checkup_$1/
RedirectMatch 301 ^/introduction_([2-5])/?$ /introduction/introduction_$1/
RedirectMatch 301 ^/introduction/introduction/?$ /introduction/

# 旧WordPressの管理画面やログインなどへのアクセスは、404にする（静的化したため、不要）。
# ただし、/kdhs/ の中の同名のファイルは、対象外（先頭が /wp-... のものだけ）。
RedirectMatch 404 ^/(wp-admin|wp-login\.php|xmlrpc\.php|wp-json|wp-config\.php|wp-cron\.php|wp-includes|readme\.html|license\.txt)
